915  dlsdk_status registerHotplugCallback(const CallbackFn& cb, CallbackHandle* outHandle)
916  {
917  static auto cppCallback = [](dlsdk_hotplug_event_data* data, void* userData) {
918  HotplugEventData eventData;
919  eventData.version = data->version;
920  eventData.event = data->event;
921  eventData.device = *reinterpret_cast<DeviceHandle*>(&data->device);
922  eventData.device_id = data->device_id ? std::string(data->device_id) : std::string();
923  eventData.display = *reinterpret_cast<DisplayHandle*>(&data->display);
924  eventData.display_id = data->display_id ? std::string(data->display_id) : std::string();
925 
926  CallbackContext* context = reinterpret_cast<CallbackContext*>(userData);
927  std::lock_guard<std::mutex> lock(context->mutex);
928  if (context->userCallback) {
929  context->userCallback(eventData);
930  }
931  };
932  {
933  std::lock_guard<std::mutex> lock(m_callbackContext.mutex);
934  m_callbackContext.userCallback = cb;
935  }
936  return dlsdk_register_hotplug_callback(cppCallback, &m_callbackContext, outHandle);
